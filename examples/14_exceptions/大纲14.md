## 2. 改造异常定义

在 `app/exceptions.py` 中改成：

```python
class AppBaseError(Exception):
    def __init__(
        self,
        message: str,
        error_code: str = "APP_ERROR",
    ):
        self.message = message
        self.error_code = error_code
        super().__init__(message)


class LLMServiceError(AppBaseError):
    def __init__(self, message="模型服务调用失败"):
        super().__init__(message, "LLM_SERVICE_ERROR")


class StructuredOutputError(AppBaseError):
    def __init__(self, message="结构化输出解析失败"):
        super().__init__(message, "STRUCTURED_OUTPUT_ERROR")
```

---

## 3. 在 `main.py` 注册全局异常处理

```python
from fastapi import FastAPI, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse

from app.exceptions import AppBaseError
from app.models import ErrorResponse
from app.routers.log_router import router as log_router


app = FastAPI(
    title="Java 日志分析器",
    version="0.3.0",
)


@app.exception_handler(AppBaseError)
async def app_error_handler(
    request: Request,
    exc: AppBaseError,
):
    status_code = 502

    if exc.error_code == "STRUCTURED_OUTPUT_ERROR":
        status_code = 422

    return JSONResponse(
        status_code=status_code,
        content=ErrorResponse(
            error_code=exc.error_code,
            message=exc.message,
        ).model_dump(),
    )


@app.exception_handler(RequestValidationError)
async def validation_exception_handler(
    request: Request,
    exc: RequestValidationError,
):
    return JSONResponse(
        status_code=422,
        content=ErrorResponse(
            error_code="VALIDATION_ERROR",
            message="请求参数校验失败",
        ).model_dump(),
    )


@app.exception_handler(Exception)
async def unknown_exception_handler(
    request: Request,
    exc: Exception,
):
    return JSONResponse(
        status_code=500,
        content=ErrorResponse(
            error_code="INTERNAL_SERVER_ERROR",
            message="服务内部错误",
        ).model_dump(),
    )


@app.get("/health")
def health():
    return {
        "status": "UP",
        "service": "log-analyzer",
    }


app.include_router(log_router)
```

然后把路由层中的大段 `try/except` 去掉：

```python
@router.post("/analyze", response_model=ApiResponse)
def analyze_log_api(request: LogAnalyzeRequest):
    result = analyze_log(request.log_text)

    return ApiResponse(
        success=True,
        data=result,
        message="分析成功",
    )
```

这样路由层更接近 Java Controller：

```text
接收参数
→ 调用 Service
→ 返回结果
```

## 4. 给请求增加 request_id

在 `log_router.py` 中：

```python
import uuid

from fastapi import APIRouter

from app.models import ApiResponse, LogAnalyzeRequest
from app.services.log_service import analyze_log


router = APIRouter(
    prefix="/api/v1/logs",
    tags=["日志分析"],
)


@router.post("/analyze", response_model=ApiResponse)
def analyze_log_api(request: LogAnalyzeRequest):
    request_id = str(uuid.uuid4())

    result = analyze_log(request.log_text)

    return ApiResponse(
        success=True,
        data=result,
        message=f"分析成功，request_id={request_id}",
    )
```

今天先把 `request_id` 放在消息里即可。
