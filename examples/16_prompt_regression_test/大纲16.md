
代码：

```python
from app.models import ErrorAnalysis


def score_result(
    result: ErrorAnalysis,
    expected_keywords: list[str],
) -> dict:
    text = " ".join(
        [
            result.error_type,
            *result.possible_causes,
            *result.suggestions,
        ]
    )

    keyword_hits = sum(
        1 for keyword in expected_keywords
        if keyword.lower() in text.lower()
    )

    score = 0

    if result.error_type.strip():
        score += 20

    if len(result.possible_causes) >= 2:
        score += 20

    if len(result.suggestions) >= 2:
        score += 20

    if len(result.evidence) >= 1:
        score += 20

    if keyword_hits >= 1:
        score += 20

    return {
        "score": score,
        "keyword_hits": keyword_hits,
        "passed": score >= 80,
    }
```

今天先不追求复杂的语义评估，只做以下检查：

* 字段是否有值
* 原因和建议数量是否足够
* 是否提到预期关键词
* 是否有证据

# 六、支持切换 Prompt 版本

修改：

```text
app/services/log_service.py
```

让 Service 支持显式传入 Prompt 版本。

```python
from app.prompts import PROMPT_REGISTRY
from config import settings


def get_prompt_config(prompt_version: str | None = None) -> dict:
    version = prompt_version or settings.prompt_version
    prompt_config = PROMPT_REGISTRY.get(version)

    if prompt_config is None:
        raise ValueError(f"不存在的 Prompt 版本：{version}")

    return prompt_config


def analyze_log(
    log_text: str,
    prompt_version: str | None = None,
) -> ErrorAnalysis:
    prompt_config = get_prompt_config(prompt_version)

    try:
        response = client.beta.chat.completions.parse(
            model=settings.model_name,
            messages=[
                {
                    "role": "system",
                    "content": prompt_config["system"],
                },
                {
                    "role": "user",
                    "content": prompt_config["user_template"].format(
                        log_text=log_text
                    ),
                },
            ],
            response_format=ErrorAnalysis,
            temperature=settings.temperature,
            max_tokens=settings.max_tokens,
        )
    except Exception as exc:
        raise LLMServiceError("模型调用失败") from exc

    message = response.choices[0].message

    if getattr(message, "refusal", None):
        raise StructuredOutputError(
            f"模型拒答：{message.refusal}"
        )

    if message.parsed is None:
        raise StructuredOutputError(
            "模型没有返回可解析的结构化结果"
        )

    return message.parsed
```

# 七、编写 Prompt 回归测试脚本

新建：

```text
app/evaluation/run_eval.py
```

代码：

```python
import json
from datetime import datetime

from app.evaluation.dataset import TEST_CASES
from app.evaluation.scorer import score_result
from app.services.log_service import analyze_log


def evaluate_prompt(prompt_version: str) -> dict:
    results = []
    total_score = 0
    passed_count = 0

    for case in TEST_CASES:
        try:
            result = analyze_log(
                case["log_text"],
                prompt_version=prompt_version,
            )

            score_info = score_result(
                result,
                case["expected_keywords"],
            )

            total_score += score_info["score"]

            if score_info["passed"]:
                passed_count += 1

            results.append(
                {
                    "case_id": case["id"],
                    "success": True,
                    "score": score_info["score"],
                    "passed": score_info["passed"],
                    "result": result.model_dump(),
                }
            )

        except Exception as exc:
            results.append(
                {
                    "case_id": case["id"],
                    "success": False,
                    "score": 0,
                    "passed": False,
                    "error": str(exc),
                }
            )

    return {
        "prompt_version": prompt_version,
        "model_name": settings.model_name,
        "run_at": datetime.now().isoformat(),
        "total_cases": len(TEST_CASES),
        "passed_count": passed_count,
        "average_score": (
            total_score / len(TEST_CASES)
            if TEST_CASES
            else 0
        ),
        "results": results,
    }


if __name__ == "__main__":
    from config import settings

    report = {
        "v1": evaluate_prompt("v1"),
        "v2": evaluate_prompt("v2"),
    }

    print(
        json.dumps(
            report,
            ensure_ascii=False,
            indent=2,
        )
    )
```

运行：

```bash
python -m app.evaluation.run_eval
```

如果你的目录导入路径不同，按照当前项目结构进行调整。

# 十一、今天的核心理解

今天真正要掌握的不是这几个 Python 文件，而是下面这个思路：

```text
修改 Prompt
    ↓
使用固定测试集
    ↓
批量调用模型
    ↓
按照固定规则评分
    ↓
比较 v1 / v2
    ↓
发现回归问题
    ↓
决定是否接受新 Prompt
```

也就是说：

> **Prompt 不应该靠“感觉不错”来判断，而应该逐渐进入工程化测试流程。**

这也是从：

```text
调用 LLM
```

进一步走向：

```text
构建可维护的 AI 应用
```

的重要一步。
